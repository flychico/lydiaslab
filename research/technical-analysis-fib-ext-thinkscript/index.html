# FIB_EXT - the desk’s 3-point Fibonacci Extension (coefficients 0.5, 0.786, 1.382, 1.618, 2, 2.382, 2.618, 3)
# Works on ANY symbol and ANY timeframe (daily, weekly, monthly...).
# AUTO: looks back "lookback" bars on the current chart, finds the highest high and lowest low.
#   If the low came first  -> up swing:   A = low, B = high, C = lowest low since B   -> levels project UP
#   If the high came first -> down swing: A = high, B = low, C = highest high since B -> levels project DOWN
# OVERRIDE: type a date as YYYYMMDD in aDate / bDate / cDate to pin that point (0 = auto).
# level = C + coefficient x (B - A), same as thinkorswim's Fibonacci Extension drawing.
declare upper;
input lookback = 100;
input aDate = 0;
input bDate = 0;
input cDate = 0;
input showExt = yes;
input showPoints = yes;
input showBubbles = yes;

def bn = BarNumber();
def d = GetYYYYMMDD();
def lastBar = HighestAll(if !IsNaN(close) then bn else 0);
def inWin = bn > lastBar - lookback;
def H = HighestAll(if inWin then high else Double.NaN);
def L = LowestAll(if inWin then low else Double.NaN);
def hBar = HighestAll(if inWin and high == H then bn else Double.NaN);
def lBar = HighestAll(if inWin and low == L then bn else Double.NaN);
def autoUp = lBar < hBar;

def aBarOv = LowestAll(if aDate > 0 and d >= aDate then bn else Double.NaN);
def bBarOv = LowestAll(if bDate > 0 and d >= bDate then bn else Double.NaN);
def aBar = if aDate > 0 and !IsNaN(aBarOv) then aBarOv else (if autoUp then lBar else hBar);
def bBar = if bDate > 0 and !IsNaN(bBarOv) then bBarOv else (if autoUp then hBar else lBar);

def aLow = HighestAll(if bn == aBar then low else Double.NaN);
def aHigh = HighestAll(if bn == aBar then high else Double.NaN);
def bLow = HighestAll(if bn == bBar then low else Double.NaN);
def bHigh = HighestAll(if bn == bBar then high else Double.NaN);
def up = if aDate > 0 or bDate > 0 then bHigh - aLow >= aHigh - bLow else autoUp;
def A = if up then aLow else aHigh;
def B = if up then bHigh else bLow;

def afterB = bn > bBar;
def pullLo = LowestAll(if afterB then low else Double.NaN);
def pullHi = HighestAll(if afterB then high else Double.NaN);
def cBarOv = LowestAll(if cDate > 0 and d >= cDate then bn else Double.NaN);
def cBarAuto = if up then HighestAll(if afterB and low == pullLo then bn else Double.NaN) else HighestAll(if afterB and high == pullHi then bn else Double.NaN);
def cBar = if cDate > 0 and !IsNaN(cBarOv) then cBarOv else (if IsNaN(cBarAuto) then bBar else cBarAuto);
def cLow = HighestAll(if bn == cBar then low else Double.NaN);
def cHigh = HighestAll(if bn == cBar then high else Double.NaN);
def C = if cBar == bBar then B else (if up then cLow else cHigh);

plot X0_5 = if showExt and bn >= cBar then C + 0.5 * (B - A) else Double.NaN;
plot X0_786 = if showExt and bn >= cBar then C + 0.786 * (B - A) else Double.NaN;
plot X1_382 = if showExt and bn >= cBar then C + 1.382 * (B - A) else Double.NaN;
plot X1_618 = if showExt and bn >= cBar then C + 1.618 * (B - A) else Double.NaN;
plot X2 = if showExt and bn >= cBar then C + 2 * (B - A) else Double.NaN;
plot X2_382 = if showExt and bn >= cBar then C + 2.382 * (B - A) else Double.NaN;
plot X2_618 = if showExt and bn >= cBar then C + 2.618 * (B - A) else Double.NaN;
plot X3 = if showExt and bn >= cBar then C + 3 * (B - A) else Double.NaN;
X0_5.SetDefaultColor(Color.WHITE); X0_5.SetStyle(Curve.SHORT_DASH); X0_5.SetLineWeight(1);
X0_786.SetDefaultColor(CreateColor(230, 185, 0)); X0_786.SetStyle(Curve.SHORT_DASH); X0_786.SetLineWeight(1);
X1_382.SetDefaultColor(Color.CYAN); X1_382.SetStyle(Curve.SHORT_DASH); X1_382.SetLineWeight(1);
X1_618.SetDefaultColor(CreateColor(230, 185, 0)); X1_618.SetStyle(Curve.SHORT_DASH); X1_618.SetLineWeight(1);
X2.SetDefaultColor(Color.WHITE); X2.SetStyle(Curve.SHORT_DASH); X2.SetLineWeight(1);
X2_382.SetDefaultColor(Color.CYAN); X2_382.SetStyle(Curve.SHORT_DASH); X2_382.SetLineWeight(1);
X2_618.SetDefaultColor(CreateColor(230, 185, 0)); X2_618.SetStyle(Curve.SHORT_DASH); X2_618.SetLineWeight(1);
X3.SetDefaultColor(Color.WHITE); X3.SetStyle(Curve.SHORT_DASH); X3.SetLineWeight(1);

AddChartBubble(showPoints and bn == aBar, if up then low else high, "A", Color.GRAY, !up);
AddChartBubble(showPoints and bn == bBar, if up then high else low, "B", Color.GRAY, up);
AddChartBubble(showPoints and bn == cBar and cBar != bBar, if up then low else high, "C", Color.GRAY, !up);

def isLast = !IsNaN(close) and IsNaN(close[-1]);
AddChartBubble(showBubbles and showExt and isLast, X0_5, "0.5 " + AsText(X0_5), Color.WHITE, yes);
AddChartBubble(showBubbles and showExt and isLast, X0_786, "0.786 " + AsText(X0_786), CreateColor(230, 185, 0), yes);
AddChartBubble(showBubbles and showExt and isLast, X1_382, "1.382 " + AsText(X1_382), Color.CYAN, yes);
AddChartBubble(showBubbles and showExt and isLast, X1_618, "1.618 " + AsText(X1_618), CreateColor(230, 185, 0), yes);
AddChartBubble(showBubbles and showExt and isLast, X2, "2 " + AsText(X2), Color.WHITE, yes);
AddChartBubble(showBubbles and showExt and isLast, X2_382, "2.382 " + AsText(X2_382), Color.CYAN, yes);
AddChartBubble(showBubbles and showExt and isLast, X2_618, "2.618 " + AsText(X2_618), CreateColor(230, 185, 0), yes);
AddChartBubble(showBubbles and showExt and isLast, X3, "3 " + AsText(X3), Color.WHITE, yes);
AddLabel(yes, "Fib ext " + (if up then "UP" else "DOWN") + "  A " + AsText(A) + "  B " + AsText(B) + "  C " + AsText(C) + (if aDate > 0 or bDate > 0 or cDate > 0 then "  (dates pinned)" else "  (auto, last " + lookback + " bars)"), CreateColor(230, 185, 0));
